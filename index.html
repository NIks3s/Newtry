<script>

// BUSINESS WHATSAPP NUMBER

// Replace with your number including country code.

// Example: 919876543210

const BUSINESS_WHATSAPP = "91XXXXXXXXXX";

const makhana = [

  "Peri Peri",

  "Cheddar Cheese",

  "Magic Masala",

  "Cream & Onion",

  "Salt & Pepper",

  "Tomato"

];

const chips = [

  "Bhel Chaska",

  "Peri Peri",

  "Jalapeno",

  "Cream & Onion",

  "Cheese & Herbs"

];

const $ = id => document.getElementById(id);

let selectedM = [];

let selectedC = [];

function limits() {

  const mix = document.querySelector(

    'input[name="mix"]:checked'

  ).value;

  return mix === "3+3" ? [3,3] : [4,2];

}

function renderGroup(items, container, kind) {

  const [ml,cl] = limits();

  const max = kind === "m" ? ml : cl;

  const arr = kind === "m" ? selectedM : selectedC;

  $(container).innerHTML = "";

  items.forEach(name => {

    const checked = arr.includes(name);

    const disabled = !checked && arr.length >= max;

    const label = document.createElement("label");

    label.className = "flavour" + (disabled ? " disabled" : "");

    const input = document.createElement("input");

    input.type = "checkbox";

    input.checked = checked;

    input.disabled = disabled;

    input.addEventListener("change", () => {

      const selection = kind === "m" ? selectedM : selectedC;

      if (input.checked) {

        selection.push(name);

      } else {

        selection.splice(selection.indexOf(name), 1);

      }

      update();

    });

    const span = document.createElement("span");

    span.textContent = name;

    label.append(input,span);

    $(container).append(label);

  });

}

function update() {

  const [ml,cl] = limits();

  selectedM = selectedM.slice(0,ml);

  selectedC = selectedC.slice(0,cl);

  renderGroup(makhana,"mChoices","m");

  renderGroup(chips,"cChoices","c");

  $("mCount").textContent = `${selectedM.length} / ${ml}`;

  $("cCount").textContent = `${selectedC.length} / ${cl}`;

  $("mHint").textContent =

    `Choose exactly ${ml} makhana flavours.`;

  $("cHint").textContent =

    `Choose exactly ${cl} chips flavours.`;

  $("mSummary").textContent =

    selectedM.join(", ") || "Not selected";

  $("cSummary").textContent =

    selectedC.join(", ") || "Not selected";

  const valid =

    selectedM.length === ml &&

    selectedC.length === cl &&

    selectedM.length + selectedC.length === 6 &&

    $("name").value.trim() &&

    $("phone").value.trim() &&

    $("address").value.trim();

  $("send").disabled = !valid;

}

// Change hamper combination

document.querySelectorAll('input[name="mix"]').forEach(input => {

  input.addEventListener("change", () => {

    selectedM = [];

    selectedC = [];

    update();

  });

});

// Update button when customer details change

["name","phone","address"].forEach(id => {

  $(id).addEventListener("input",update);

});

// WHATSAPP ORDER FUNCTION

$("send").addEventListener("click", () => {

  const name = $("name").value.trim();

  const phone = $("phone").value.trim();

  const address = $("address").value.trim();

  const note = $("note").value.trim();

  const mix = document.querySelector(

    'input[name="mix"]:checked'

  ).value;

  if (!name || !phone || !address) {

    alert("Please complete all required details.");

    return;

  }

  if (

    selectedM.length + selectedC.length !== 6 ||

    selectedM.length !== limits()[0] ||

    selectedC.length !== limits()[1]

  ) {

    alert("Please select the correct hamper combination.");

    return;

  }

  if (BUSINESS_WHATSAPP === "91XXXXXXXXXX") {

    alert("Please add your business WhatsApp number in the website code first.");

    return;

  }

  const message =

`Hello Makhana Mantra!

I would like to place a customised hamper order.

CUSTOMER DETAILS

Name: ${name}

Mobile: ${phone}

Delivery Address: ${address}

HAMPER DETAILS

Combination: ${mix === "4+2" ? "4 Makhana + 2 Chips" : "3 Makhana + 3 Chips"}

Total Products: 6

MAKHANA 40g

${selectedM.join(", ")}

MAKHANA CHIPS 90g

${selectedC.join(", ")}

Hamper Price: ₹850

Gift Message / Instructions:

${note || "None"}

Please confirm my order. Thank you!`;

  const whatsappURL =

    "https://wa.me/" +

    BUSINESS_WHATSAPP +

    "?text=" +

    encodeURIComponent(message);

  window.open(whatsappURL,"_blank");

});

update();

</script>
